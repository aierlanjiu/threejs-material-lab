function createWeaveNormal() {
  const size = 256;
  const c = document.createElement('canvas');
  c.width = c.height = size;
  const ctx = c.getContext('2d');
  ctx.fillStyle = '#8080ff';
  ctx.fillRect(0, 0, size, size);
  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const warp = Math.sin(x * 0.35) * 16;
      const weft = Math.sin(y * 0.35) * 16;
      const v = 128 + warp + weft;
      ctx.fillStyle = `rgb(${Math.max(70, Math.min(185, v))},${Math.max(70, Math.min(185, v))},255)`;
      ctx.fillRect(x, y, 1, 1);
    }
  }
  const t = new THREE.CanvasTexture(c);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.repeat.set(4, 4);
  return t;
}

function drawLiveAvatar(canvas, avatarKey, time, lookX, lookY, isBlinking, emote, bassEnergy = 0) {
  const ctx = canvas.getContext('2d');
  const size = canvas.width;
  ctx.clearRect(0, 0, size, size);

  const item = ONEWORKS_AVATAR_CONFIGS[avatarKey] || ONEWORKS_AVATAR_CONFIGS.cloud;
  const pad = size * 0.04;
  const cardSize = size - pad * 2;
  const radius = cardSize * 0.28;

  ctx.save();
  ctx.beginPath();
  ctx.roundRect(pad, pad, cardSize, cardSize, radius);
  ctx.clip();
  ctx.fillStyle = item.bg;
  ctx.fillRect(pad, pad, cardSize, cardSize);

  const breathY = Math.sin(time * 3) * (size * 0.012) - (bassEnergy * size * 0.035);
  const breathScale = 1.0 + (Math.cos(time * 3) * 0.01) + (bassEnergy * 0.04);
  const cx = size / 2;
  const cy = size * 0.62 + breathY;

  const eyeGazeX = lookX * (size * 0.035);
  const eyeGazeY = lookY * (size * 0.025);

  let eyeHeightFactor = isBlinking ? 0.08 : 1.0;
  if (emote === 'wink') eyeHeightFactor = 0.08;

  ctx.shadowColor = 'rgba(0, 0, 0, 0.16)';
  ctx.shadowBlur = 12;
  ctx.shadowOffsetY = 6;
  ctx.fillStyle = item.bodyColor;

  if (item.type === 'cloud') {
    ctx.save();
    ctx.translate(cx, cy);
    ctx.scale(breathScale, breathScale);
    ctx.translate(-cx, -cy);
    ctx.beginPath();
    ctx.moveTo(pad - 10, size);
    ctx.lineTo(pad - 10, cy + size * 0.1);
    ctx.quadraticCurveTo(cx - size * 0.35, cy - size * 0.15, cx - size * 0.18, cy - size * 0.12);
    ctx.quadraticCurveTo(cx, cy - size * 0.42, cx + size * 0.28, cy - size * 0.18);
    ctx.quadraticCurveTo(size + 10, cy - size * 0.1, size + 10, size);
    ctx.closePath();
    ctx.fill();
    ctx.restore();

    ctx.shadowBlur = 0;
    ctx.fillStyle = '#050608';
    const eyeW = size * 0.07;
    const baseEyeH = size * 0.14;
    const eyeH = baseEyeH * (emote === 'happy' ? 0.15 : eyeHeightFactor);
    const eyeRx = eyeW / 2;
    const eyeLeftX = cx + size * 0.02 + eyeGazeX;
    const eyeRightX = cx + size * 0.16 + eyeGazeX;
    const eyeY = cy - size * 0.02 + eyeGazeY + (baseEyeH - eyeH) / 2;

      ctx.strokeStyle = '#050608';
      ctx.lineCap = 'round';
      ctx.beginPath(); ctx.arc(eyeLeftX + eyeW/2, eyeY + eyeH, eyeW, Math.PI * 1.1, Math.PI * 1.9); ctx.stroke();

    } else {
      ctx.beginPath(); ctx.roundRect(eyeLeftX, eyeY, eyeW, eyeH, eyeRx); ctx.fill();
      ctx.beginPath(); ctx.roundRect(eyeRightX, eyeY, eyeW, (emote === 'wink' ? baseEyeH : eyeH), eyeRx); ctx.fill();
    }
  } else if (item.type === 'rabbit') {
    const earBounce = Math.sin(time * 4) * 0.05 + (bassEnergy * 0.1);

// 鏅鸿兘楂樺姣旀枃瀛楄壊褰╁喅绛栬绠?function getSmartContrastTextColor(baseColorHex, userPreference = 'auto') {
  if (userPreference === 'dark') {
    return {
      fill: '#090d16',
      shadow: 'rgba(255,255,255,0.45)',
      blur: 8,
      stroke: 'rgba(255,255,255,0.25)',
      strokeWidth: 2
    };
  }
  if (userPreference === 'white') {
    return {
      fill: '#ffffff',
      shadow: 'rgba(0,0,0,0.92)',
      blur: 22,
      stroke: 'rgba(0,0,0,0.5)',
      strokeWidth: 4
    };
  }
  if (userPreference === 'gold') {
    return {
      fill: '#f59e0b',
      shadow: 'rgba(0,0,0,0.75)',
      blur: 16,
      stroke: '#78350f',
      strokeWidth: 3
    };
  }
  if (userPreference === 'cyan') {
    return {
      fill: '#0284c7',
      shadow: 'rgba(255,255,255,0.5)',
      blur: 12,
      stroke: 'rgba(255,255,255,0.3)',
      strokeWidth: 2
    };
  }

  const c = new THREE.Color(baseColorHex);
  const luminance = 0.299 * c.r + 0.587 * c.g + 0.114 * c.b;

  if (luminance > 0.52) {
    return {
      fill: '#090d16',
      shadow: 'rgba(255,255,255,0.55)',
      blur: 10,
      stroke: 'rgba(255,255,255,0.3)',
      strokeWidth: 2
    };
  } else {
    return {
      fill: '#ffffff',
      shadow: 'rgba(0,0,0,0.88)',
      blur: 18,
      stroke: 'rgba(0,0,0,0.4)',
      strokeWidth: 3
    };
  }
}

// 鏂囧瓧璐村浘鐢熸垚
function createTypographyTexture(char, craftType, baseColorHex) {
  const size = 512;
  const c = document.createElement('canvas');
  c.width = c.height = size;
  const ctx = c.getContext('2d');
  ctx.clearRect(0, 0, size, size);

  const style = getSmartContrastTextColor(baseColorHex, state.textColorMode);

  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.font = '900 236px -apple-system, BlinkMacSystemFont, "PingFang SC", "Hiragino Sans GB", "Microsoft YaHei", sans-serif';

  ctx.shadowColor = style.shadow;
  ctx.shadowBlur = style.blur;
  ctx.shadowOffsetY = 6;

  if (style.strokeWidth > 0) {
    ctx.lineWidth = style.strokeWidth * 2;
    ctx.strokeStyle = style.stroke;
    ctx.strokeText(char, size / 2, size / 2 + 8);
  }

  ctx.fillStyle = style.fill;
  ctx.fillText(char, size / 2, size / 2 + 8);

  const texture = new THREE.CanvasTexture(c);
  texture.colorSpace = THREE.SRGBColorSpace;
  return texture;
}

function clearMatrixGroup() {
  while (matrixGroup.children.length) {
    const obj = matrixGroup.children.pop();
    obj.traverse?.(child => {
      if (child.material) {
        const mats = Array.isArray(child.material) ? child.material : [child.material];
        mats.forEach(m => {
          if (m.map && !Object.values(liveTextureMap).includes(m.map) && m.map !== state.customImageTexture && m.map !== jadeMarblingTexture && m.map !== cloudVolumeTexture) {
            m.map.dispose();
          }
          m.dispose();
        });
      }
      if (child.geometry && child.geometry !== currentRoundedGeo) {
        child.geometry.dispose();
      }
    });
  }
}

function playTrackAtIndex(idx) {
  initAudioEngine();
  if (state.audioCtx.state === 'suspended') state.audioCtx.resume();

  state.currentTrackIndex = (idx + state.playlist.length) % state.playlist.length;
  const track = state.playlist[state.currentTrackIndex];

  state.isPlayingMusic = true;

  } else {
    startProceduralElectronicBeat();

function startProceduralElectronicBeat() {
  if (synthInterval) clearInterval(synthInterval);

  let step = 0;
  const baseBpm = 124;
  const bpm = baseBpm * state.speedMultiplier;
  const stepTime = (60 / bpm) / 4 * 1000;

  synthInterval = setInterval(() => {
    if (!state.isPlayingMusic) return;
    const now = state.audioCtx.currentTime;

    if (step % 4 === 0) {
      const osc = state.audioCtx.createOscillator();
      const gain = state.audioCtx.createGain();
      osc.frequency.setValueAtTime(150, now);
      osc.frequency.exponentialRampToValueAtTime(36, now + 0.14);
      gain.gain.setValueAtTime(1.1, now);
      gain.gain.exponentialRampToValueAtTime(0.01, now + 0.16);
      osc.connect(gain);
      gain.connect(state.analyser);
      gain.connect(state.audioCtx.destination);
      osc.start(now);
      osc.stop(now + 0.16);
      state.beatCount++;
    }

    if (step % 8 === 4) {
      const noise = state.audioCtx.createBufferSource();
      const buffer = state.audioCtx.createBuffer(1, state.audioCtx.sampleRate * 0.12, state.audioCtx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (data.length * 0.28));
      noise.buffer = buffer;
      const gain = state.audioCtx.createGain();
      gain.gain.setValueAtTime(0.7, now);
      gain.gain.exponentialRampToValueAtTime(0.01, now + 0.13);
      noise.connect(gain);
      gain.connect(state.analyser);
      gain.connect(state.audioCtx.destination);
      noise.start(now);
      noise.stop(now + 0.14);
    }

    if (step % 2 === 1) {
      const osc = state.audioCtx.createOscillator();
      const gain = state.audioCtx.createGain();
      const bassProgression = [110, 110, 146.83, 164.81, 130.81, 130.81, 164.81, 196.00];
      const note = bassProgression[(step / 2 | 0) % bassProgression.length];
      osc.frequency.setValueAtTime(note, now);
      gain.gain.setValueAtTime(0.4, now);
      gain.gain.exponentialRampToValueAtTime(0.01, now + 0.09);
      osc.connect(gain);
      gain.connect(state.analyser);
      gain.connect(state.audioCtx.destination);
      osc.start(now);
      osc.stop(now + 0.10);
    }

    step = (step + 1) % 32;
  }, stepTime);
}

function updateMasterChoreography() {
  if (!state.isPlayingMusic) return;

  const interval = state.morphIntervalBeats || 16;
  const moveIdx = Math.floor(state.beatCount / interval) % CHOREOGRAPHY_MOVEMENTS.length;
  $('#movementBadge').textContent = `涔愮珷 ${moveIdx + 1}/${CHOREOGRAPHY_MOVEMENTS.length}`;

  if (state.isChoreographyShow && moveIdx !== lastMovementIdx) {
    lastMovementIdx = moveIdx;
    const mov = CHOREOGRAPHY_MOVEMENTS[moveIdx];

    state.isIntroPlaying = mov.isIntro && state.enableTitleIntro;
    state.avatarDist = mov.dist;
    state.gridCols = mov.cols;
    state.gridRows = mov.rows;
    state.gridLayers = mov.layers;

    ui.gridRows.value = mov.rows;
    ui.gridLayers.value = mov.layers;

    rebuildMatrix();
  }

  const maxDim = Math.max(state.gridCols, state.gridRows, state.gridLayers);
  let baseCamDist = 4.2 + maxDim * 2.2;

  if (state.enableDynamicFov) {
    if (state.fovMode === 'bass') {

      const breath = Math.sin(state.time * 2.5) * (state.fovPulseAmp * 0.22);
      targetFov = baseFov - kickImpulse + breath;
    } else if (state.fovMode === 'dolly') {

      baseCamDist = baseCamDist * (Math.tan(fovRad1) / Math.tan(fovRad2));
    } else if (state.fovMode === 'sine') {

      const beatCycle = Math.floor(state.beatCount / 4) % 2;

  const baseBloom = +ui.bloom.value;
  const baseLight = +ui.light.value;
  const baseDispersion = +ui.dispersion.value;
  const baseEnv = +ui.envIntensity.value;
  const baseRoughness = +ui.roughness.value;

  const dynamicDispersion = Math.min(0.50, baseDispersion + (state.bassEnergy * 0.18 + state.trebleEnergy * 0.08));

  // 绮楃硻搴︺€両OR涓庡厜瀛﹀帤搴﹀姩鎬佸叡鎸?  const dynamicRoughness = Math.min(0.08, baseRoughness + (state.midEnergy * 0.03));

        if (m.transmission > 0) {
          m.ior = dynamicIor;
          m.thickness = dynamicThickness;
          m.dispersion = dynamicDispersion;
        }
      }
    });
  });
}

    for (let i = 4; i < 20; i++) midSum += state.audioDataArray[i];
    for (let i = 20; i < 48; i++) trebleSum += state.audioDataArray[i];

    if (state.bassEnergy > 0.42 && (state.bassEnergy - state.lastBassEnergy) > 0.08 && (now - state.lastBeatTime) > minBeatInterval) {
      state.beatCount++;
      state.lastBeatTime = now;
    }

    matrixGroup.children.forEach(unit => {

        if (state.wavePattern === 'ripple') {
          // 1. 鍚屽績鍦嗙幆姘存尝
          const phase = state.waveAccumulator * 4.5 - (uData.distFromCenter * 1.8);
          const waveAmp = (state.bassEnergy * 0.45 + state.midEnergy * 0.2) * amp;

        } else if (state.wavePattern === 'glitch') {
          // 6. 璧涘崥闅忔満鑴夊啿
          if (Math.random() < state.bassEnergy * 0.35) {
            targetY = uData.baseY + (Math.random() * 0.7 - 0.35) * amp;
          }
        }

        unit.position.y += (targetY - unit.position.y) * 0.28;
        unit.rotation.x += (rotX - unit.rotation.x) * 0.25;
        unit.rotation.z += (rotZ - unit.rotation.z) * 0.25;
      } else {
        unit.position.y += (uData.baseY - unit.position.y) * 0.1;
        unit.rotation.x *= 0.9;
        unit.rotation.z *= 0.9;
        unit.scale.lerp(new THREE.Vector3(1, 1, 1), 0.1);
      }
    });
  }

  if (state.enableBlink) {
    const blinkCycle = state.time % 3.6;
    isBlinking = blinkCycle < 0.16;
  }
  const lookX = state.gazeFollow ? state.mouseNormX : 0;
  const lookY = state.gazeFollow ? state.mouseNormY : 0;

  let currentEmote = null;

    currentEmote = state.activeEmote;
  } else if (state.bassEnergy > 0.6) {

    AVATAR_KEYS.forEach(key => {
      const cvs = liveCanvasMap[key];
      drawLiveAvatar(cvs, key, state.time, lookX, lookY, isBlinking, currentEmote, state.bassEnergy);
      liveTextureMap[key].needsUpdate = true;
    });

    const activeThumb = $(`#thumb_${state.selectedAvatar}`);
    if (activeThumb) {
      activeThumb.width = activeThumb.height = 88;
      drawLiveAvatar(activeThumb, state.selectedAvatar, state.time, lookX, lookY, isBlinking, currentEmote, state.bassEnergy);
    }
  }

  if (ui.material.value === 'cloud' && cloudVolumeTexture) {
    cloudVolumeTexture.offset.x = (state.time * 0.015) % 1;
    cloudVolumeTexture.offset.y = (state.time * 0.01) % 1;
  }

  syncSliderOutputs();
  composer.render();
  requestAnimationFrame(animate);
}

    else if (type === 'top') camera.position.set(0, targetCamDist * 1.2, 0.01);
    else if (type === 'iso') camera.position.set(targetCamDist * 0.65, targetCamDist * 0.55, targetCamDist * 0.75);
    controls.target.set(0, 0, 0);
    controls.update();
  });
});
